use mysql;
# Python_db : DB를 생성하겠다.
create database python_db;
# 생성 후 DB 목록 다시 확인하기
show databases;
# user는 python, password는 python 계정을 생성하겠다.
create user 'python'@'%' identified by 'python';
# Python 계정에게 Python_db 권한을 부여하겠다.
grant all privileges on python_db.* to 'python'@'%';
# 권한 부여를 DB에 즉시 적용하기
flush privileges;
# user가 생성되었는지 확인하기
select user, host from user;
exit;
